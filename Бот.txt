def init_db():
    conn = sqlite3.connect('max_bot.db')
    cursor = conn.cursor()
    
    cursor.executescript('''
        CREATE TABLE IF NOT EXISTS users (
            user_id INTEGER PRIMARY KEY, username TEXT, balance REAL DEFAULT 0,
            total_earned REAL DEFAULT 0, total_withdrawn REAL DEFAULT 0,
            hold_balance REAL DEFAULT 0, total_orders INTEGER DEFAULT 0,
            success_orders INTEGER DEFAULT 0, cancelled_orders INTEGER DEFAULT 0,
            registration_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            last_activity TIMESTAMP DEFAULT CURRENT_TIMESTAMP, is_blocked BOOLEAN DEFAULT 0
        );
        
        CREATE TABLE IF NOT EXISTS code_queue (
            id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER, phone_number TEXT,
            status TEXT DEFAULT 'waiting', created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            taken_by INTEGER, taken_at TIMESTAMP, code_sent BOOLEAN DEFAULT 0,
            code_value TEXT, attempts INTEGER DEFAULT 0, completed_at TIMESTAMP,
            failed_at TIMESTAMP, problem_at TIMESTAMP
        );
        
        CREATE TABLE IF NOT EXISTS qr_queue (
            id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER,
            status TEXT DEFAULT 'waiting', created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
            taken_by INTEGER, taken_at TIMESTAMP, qr_sent BOOLEAN DEFAULT 0,
            qr_file_id TEXT, qr_message_id INTEGER, attempts INTEGER DEFAULT 0,
            completed INTEGER DEFAULT 0, completed_at TIMESTAMP, failed_at TIMESTAMP,
            problem_at TIMESTAMP
        );
        
        CREATE TABLE IF NOT EXISTS transactions (
            id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER, amount REAL,
            type TEXT, description TEXT, status TEXT DEFAULT 'completed',
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        );
        
        CREATE TABLE IF NOT EXISTS withdraw_requests (
            id INTEGER PRIMARY KEY AUTOINCREMENT, user_id INTEGER, amount REAL,
            status TEXT DEFAULT 'pending', admin_id INTEGER, check_link TEXT,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, processed_at TIMESTAMP
        );
        
        CREATE TABLE IF NOT EXISTS settings (
            key TEXT PRIMARY KEY, value TEXT, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        );
    ''')
    
    for key, value in [('code_price','4.00'),('qr_price','4.00'),('min_withdraw','1.00'),('hold_time','300')]:
        cursor.execute('INSERT OR IGNORE INTO settings (key, value) VALUES (?,?)', (key, value))
    
    conn.commit()
    conn.close()

def is_admin(uid): return uid in ADMIN_IDS

def get_setting(key, default=None):
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    c.execute('SELECT value FROM settings WHERE key=?', (key,))
    r = c.fetchone()
    conn.close()
    return r[0] if r else default

def update_setting(key, value):
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    c.execute('INSERT OR REPLACE INTO settings (key,value,updated_at) VALUES (?,?,CURRENT_TIMESTAMP)', (key, str(value)))
    conn.commit()
    conn.close()

def check_subscription(uid):

def get_queue_position(uid, qtype='code'):
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    if qtype == 'code':
        c.execute('SELECT COUNT(*) FROM code_queue WHERE status="waiting" AND id<=(SELECT COALESCE(MIN(id),0) FROM code_queue WHERE user_id=? AND status="waiting")', (uid,))
    else:
        c.execute('SELECT COUNT(*) FROM qr_queue WHERE status="waiting" AND id<=(SELECT COALESCE(MIN(id),0) FROM qr_queue WHERE user_id=? AND status="waiting")', (uid,))
    p = c.fetchone()[0]
    conn.close()
    return p

def add_to_queue(uid, qtype='code', phone=None):
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    if qtype == 'code':
        c.execute('SELECT id FROM code_queue WHERE user_id=? AND status="waiting"', (uid,))
        if c.fetchone():
            conn.close()
            return False, "Вы уже в очереди"
        c.execute('INSERT INTO code_queue (user_id,phone_number,status) VALUES (?,?,"waiting")', (uid, phone))
    else:
        c.execute('SELECT id FROM qr_queue WHERE user_id=? AND status="waiting"', (uid,))
        if c.fetchone():
            conn.close()
            return False, "Вы уже в очереди QR"
        c.execute('INSERT INTO qr_queue (user_id,status) VALUES (?,"waiting")', (uid,))
    conn.commit()
    conn.close()
    return True, "Добавлены в очередь"

def show_profile(message):
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    c.execute('SELECT * FROM users WHERE user_id=?', (message.chat.id,))
    user = c.fetchone()
    
    if not user:
        show_main_menu(message.chat.id, message.message_id)
        conn.close()
        return
    
    c.execute('SELECT COUNT(*) as t, SUM(CASE WHEN status="completed" THEN 1 ELSE 0 END) as ok, SUM(CASE WHEN status="failed" THEN 1 ELSE 0 END) as fl, SUM(CASE WHEN status IN ("taken","waiting") THEN 1 ELSE 0 END) as ip FROM code_queue WHERE user_id=?', (message.chat.id,))
    cs = c.fetchone()
    c.execute('SELECT COUNT(*) as t, SUM(CASE WHEN status="completed" THEN 1 ELSE 0 END) as ok, SUM(CASE WHEN status="failed" THEN 1 ELSE 0 END) as fl, SUM(CASE WHEN status IN ("taken","waiting") THEN 1 ELSE 0 END) as ip FROM qr_queue WHERE user_id=?', (message.chat.id,))
    qs = c.fetchone()
    
    total = (cs[0] or 0) + (qs[0] or 0)
    ok = (cs[1] or 0) + (qs[1] or 0)
    fl = (cs[2] or 0) + (qs[2] or 0)
    ip = (cs[3] or 0) + (qs[3] or 0)
    rate = (ok/total*100) if total > 0 else 0
    
    c.execute('SELECT amount, description FROM transactions WHERE user_id=? ORDER BY created_at DESC LIMIT 1', (message.chat.id,))
    lt = c.fetchone()
    
    c.execute('SELECT COUNT(*) FROM withdraw_requests WHERE user_id=? AND status="pending"', (message.chat.id,))
    pw = c.fetchone()[0]
    conn.close()
    
    text = (

# ==================== ХОЛД И НАЧИСЛЕНИЕ ====================
def process_hold_payment(user_id, order_id, amount, order_type='code'):
    try:
        conn = sqlite3.connect('max_bot.db')
        c = conn.cursor()
        
        if order_type == 'code':
            c.execute('SELECT status FROM code_queue WHERE id=?', (order_id,))
        else:
            c.execute('SELECT status FROM qr_queue WHERE id=?', (order_id,))
        
        order = c.fetchone()
        
        if order and order[0] == 'completed':
            c.execute('UPDATE users SET balance=balance+?, total_earned=total_earned+? WHERE user_id=?', (amount, amount, user_id))
            c.execute('INSERT INTO transactions (user_id, amount, type, description) VALUES (?,?,"income",?)', (user_id, amount, f'Оплата за заказ #{order_id}'))
            c.execute('UPDATE users SET success_orders=success_orders+1 WHERE user_id=?', (user_id,))
            conn.commit()
            
            bot.send_message(user_id,
                f"💰 *СРЕДСТВА ЗАЧИСЛЕНЫ!*\n━━━━━━━━━━━━━━━━━━━\n\n💵 *Сумма:* ${amount:.2f}\n📋 *Заказ:* #{order_id}\n\n✅ Холд завершен\nСредства доступны на балансе\n\n━━━━━━━━━━━━━━━━━━━\n🙏 Спасибо за сотрудничество!",
                parse_mode='Markdown')
        conn.close()
    except Exception as e:
        logger.error(f"Ошибка начисления: {e}")

@bot.message_handler(func=lambda m: user_states.get(m.from_user.id,{}).get('state')=='waiting_withdraw_amount')
def process_withdraw_amount(message):
    try:
        amount = float(message.text.strip())
    except:
        bot.reply_to(message, "❌ Введите число, например: `10.50`", parse_mode='Markdown')
        return
    
    del user_states[message.from_user.id]
    
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    c.execute('SELECT balance, hold_balance FROM users WHERE user_id=?', (message.from_user.id,))
    user = c.fetchone()
    
    if not user:
        bot.send_message(message.chat.id, "❌ Ошибка")
        conn.close()
        return
    
    avail = user[0] - user[1]
    
    if amount < 1.0 or amount > avail:
        bot.send_message(message.chat.id, f"❌ Неверная сумма! Доступно: ${avail:.2f}", parse_mode='Markdown')
        conn.close()
        return
    
    c.execute('INSERT INTO withdraw_requests (user_id, amount, status) VALUES (?,?,"pending")', (message.from_user.id, amount))
    c.execute('UPDATE users SET balance=balance-?, hold_balance=hold_balance+? WHERE user_id=?', (amount, amount, message.from_user.id))
    wid = c.lastrowid
    conn.commit()
    conn.close()
    
    bot.send_message(message.chat.id,
        f"✅ *ЗАЯВКА НА ВЫВОД СОЗДАНА!*\n━━━━━━━━━━━━━━━━━━━\n\n📋 ID: #{wid}\n💳 Сумма: ${amount:.2f}\n📅 Дата: {datetime.now().strftime('%d.%m.%Y %H:%M')}\n\n⏱ *Статус: Ожидает обработки*\n\n━━━━━━━━━━━━━━━━━━━",
        parse_mode='Markdown')
    
    for aid in ADMIN_IDS:
        try:
            markup = types.InlineKeyboardMarkup()
            markup.add(types.InlineKeyboardButton("✅ Выплатить", callback_data=f'process_withdraw_{wid}'))
            bot.send_message(aid,

                parse_mode='Markdown', reply_markup=markup)

        return
    
    c.execute('UPDATE withdraw_requests SET status="completed", admin_id=?, check_link=?, processed_at=CURRENT_TIMESTAMP WHERE id=?', (message.from_user.id, link, wid))

        get_next_code(call)
    elif data == 'get_qr':
        get_next_qr(call)
    elif data.startswith('send_code_'):
        handle_send_code(call)
    elif data.startswith('skip_code_'):
        handle_skip_code(call)
    elif data.startswith('enter_code_'):
        handle_enter_code_request(call)
    elif data.startswith('code_success_'):

    elif data.startswith('qr_password_'):
        handle_qr_password(call)

        handle_qr_failed(call)

@bot.message_handler(func=lambda m: user_states.get(m.from_user.id,{}).get('state')=='admin_editing_balance')
def admin_edit_balance_exec(message):
    if not is_admin(message.from_user.id): return
    
    us = user_states[message.from_user.id]
    uid = us['user_id']
    del user_states[message.from_user.id]
    
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    c.execute('SELECT balance FROM users WHERE user_id=?', (uid,))
    u = c.fetchone()
    
    if not u:
        bot.reply_to(message, "❌ Не найден")
        conn.close()
        return
    
    old = u[0]
    inp = message.text.strip()
    
    try:
        if inp.startswith('+'):
            new = old + float(inp[1:])
        elif inp.startswith('-'):
            new = max(0, old - float(inp[1:]))
        else:
            new = float(inp)
    except:
        bot.reply_to(message, "❌ Неверный формат")
        conn.close()
        return
    
    c.execute('UPDATE users SET balance=? WHERE user_id=?', (new, uid))
    conn.commit()
    conn.close()
    
    bot.reply_to(message,
        f"✅ Баланс обновлен!\n${old:.2f} → ${new:.2f}",
        reply_markup=types.InlineKeyboardMarkup().add(types.InlineKeyboardButton("◀️ Назад", callback_data=f'admin_user_info_{uid}')))
