def get_setting(key, default=None):
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    c.execute('SELECT value FROM settings WHERE key=?', (key,))
    r = c.fetchone()
    conn.close()
    return r[0] if r else default

def update_setting(key, value):
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    c.execute('INSERT OR REPLACE INTO settings (key,value,updated_at) VALUES (?,?,CURRENT_TIMESTAMP)', (key, str(value)))
    conn.commit()
    conn.close()

def get_queue_position(uid, qtype='code'):
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    if qtype == 'code':
        c.execute('SELECT COUNT(*) FROM code_queue WHERE status="waiting" AND id<=(SELECT COALESCE(MIN(id),0) FROM code_queue WHERE user_id=? AND status="waiting")', (uid,))
    else:
        c.execute('SELECT COUNT(*) FROM qr_queue WHERE status="waiting" AND id<=(SELECT COALESCE(MIN(id),0) FROM qr_queue WHERE user_id=? AND status="waiting")', (uid,))
    p = c.fetchone()[0]
    conn.close()
    return p

def add_to_queue(uid, qtype='code', phone=None):
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    if qtype == 'code':
        c.execute('SELECT id FROM code_queue WHERE user_id=? AND status="waiting"', (uid,))
        if c.fetchone():
            conn.close()
            return False, "Вы уже в очереди"
        c.execute('INSERT INTO code_queue (user_id,phone_number,status) VALUES (?,?,"waiting")', (uid, phone))
    else:
        c.execute('SELECT id FROM qr_queue WHERE user_id=? AND status="waiting"', (uid,))
        if c.fetchone():
            conn.close()
            return False, "Вы уже в очереди QR"
        c.execute('INSERT INTO qr_queue (user_id,status) VALUES (?,"waiting")', (uid,))
    conn.commit()
    conn.close()
    return True, "Добавлены в очередь"

def show_profile(message):
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    c.execute('SELECT * FROM users WHERE user_id=?', (message.chat.id,))
    user = c.fetchone()
    if not user: show_main_menu(message.chat.id, message.message_id); conn.close(); return
    c.execute('SELECT COUNT(*) as t, SUM(CASE WHEN status="completed" THEN 1 ELSE 0 END) as ok, SUM(CASE WHEN status="failed" THEN 1 ELSE 0 END) as fl, SUM(CASE WHEN status IN ("taken","waiting") THEN 1 ELSE 0 END) as ip FROM code_queue WHERE user_id=?', (message.chat.id,))
    cs = c.fetchone()

    ok = (cs[1] or 0) + (qs[1] or 0)
    fl = (cs[2] or 0) + (qs[2] or 0)
    ip = (cs[3] or 0) + (qs[3] or 0)
    rate = (ok/total*100) if total > 0 else 0
    c.execute('SELECT amount, description FROM transactions WHERE user_id=? ORDER BY created_at DESC LIMIT 1', (message.chat.id,))
    lt = c.fetchone()
    c.execute('SELECT COUNT(*) FROM withdraw_requests WHERE user_id=? AND status="pending"', (message.chat.id,))
    pw = c.fetchone()[0]
    conn.close()
    text = (
        f"👤 **МОЙ ПРОФИЛЬ**\n\n📱 **Информация**\n• ID: `{user[0]}`\n• Username: @{user[1] or 'Не указан'}\n• Статус: {'🟢 В работе' if ip > 0 else '⚪ Ожидание'}\n\n"

    )

@bot.message_handler(func=lambda m: user_states.get(m.from_user.id,{}).get('state')=='waiting_code')
def process_code_input(message):
    code = message.text.strip()
    if not re.match(r'^\d{5,6}$', code): bot.reply_to(message, "❌ Неверный формат\n\nОтправьте 5-6 цифр кода"); return
    us = user_states[message.from_user.id]
    oid = us['order_id']
    del user_states[message.from_user.id]
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    c.execute('UPDATE code_queue SET code_value=?, code_sent=1 WHERE id=?', (code, oid))
    c.execute('SELECT taken_by, phone_number FROM code_queue WHERE id=?', (oid,))
    order = c.fetchone()
    conn.commit()
    conn.close()
    if order and order[0]:
        thread_id = None
        if hasattr(bot, 'order_threads') and oid in bot.order_threads: thread_id = bot.order_threads[oid]
        markup = types.InlineKeyboardMarkup(row_width=2)
        markup.add(types.InlineKeyboardButton("✅ Встал", callback_data=f'code_success_{oid}'), types.InlineKeyboardButton("🔐 Пароль", callback_data=f'code_password_{oid}'))
        try: bot.send_message(order[0], f"🔑 **ПОЛУЧЕН КОД**\n\n📱 Номер: `{order[1]}`\n🔢 Код: `{code}`\n📋 Заявка: #{oid}\n\n✅ Встал — вход выполнен\n🔐 Пароль — запрос пароля", parse_mode='Markdown', reply_markup=markup)
        except: pass
        if thread_id is not None:
            try: bot.send_message(GROUP_ID, f"🔑 **КОД ПОДТВЕРЖДЕНИЯ**\n\n📱 `{order[1]}`\n🔢 `{code}`\n📋 #{oid}", parse_mode='Markdown', message_thread_id=thread_id)
            except: pass
        markup = types.InlineKeyboardMarkup()
        markup.add(types.InlineKeyboardButton("◀️ В меню", callback_data='back_to_menu'))
        bot.send_message(message.chat.id, "✅ **Код отправлен обработчику**\n\n⏱ Ожидайте подтверждение...", parse_mode='Markdown', reply_markup=markup)
    else: bot.send_message(message.chat.id, "❌ Произошла ошибка")

def handle_code_success(call):
    oid = int(call.data.split('_')[2])
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    c.execute('SELECT user_id, phone_number FROM code_queue WHERE id=?', (oid,))
    order = c.fetchone()
    if order:
        price = float(get_setting('code_price','4.00'))
        ht = int(get_setting('hold_time','300'))
        c.execute('UPDATE code_queue SET status="completed", completed_at=CURRENT_TIMESTAMP WHERE id=?', (oid,))
        c.execute('UPDATE users SET total_orders=total_orders+1 WHERE user_id=?', (order[0],))
        conn.commit()
        conn.close()
        try: bot.send_message(order[0], f"🎉 **НОМЕР УСПЕШНО ВСТАЛ!**\n\n📱 `{order[1]}`\n✅ Статус: Подтверждено\n\n💰 Начисление: **${price:.2f}**\n⏱ Холд: {ht//60} мин\n\nСредства поступят через {ht//60} минут", parse_mode='Markdown')
        except: pass
        timer = threading.Timer(ht, process_hold_payment, args=[order[0], oid, price, 'code'])
        timer.start()
        if not hasattr(bot, 'hold_timers'): bot.hold_timers = {}
        bot.hold_timers[f'code_{oid}'] = {'timer': timer, 'user_id': order[0], 'amount': price}
        markup = types.InlineKeyboardMarkup()
        markup.add(types.InlineKeyboardButton("❌ Слетел", callback_data=f'code_failed_{oid}'))
        bot.edit_message_text(f"✅ **НОМЕР ВСТАЛ**\n\n📱 `{order[1]}`\n📋 #{oid}\n\n💰 ${price:.2f}\n⏱ Холд {ht//60} мин\n\n⚠️ Нажмите «Слетел» если номер упал", call.message.chat.id, call.message.message_id, parse_mode='Markdown', reply_markup=markup)
        bot.answer_callback_query(call.id, f"✅ Холд {ht//60} мин")
    else: bot.answer_callback_query(call.id, "❌ Заявка не найдена", show_alert=True)

def process_hold_payment(user_id, order_id, amount, order_type='code'):
    try:
        conn = sqlite3.connect('max_bot.db')
        c = conn.cursor()
        if order_type == 'code': c.execute('SELECT status FROM code_queue WHERE id=?', (order_id,))
        else: c.execute('SELECT status FROM qr_queue WHERE id=?', (order_id,))
        order = c.fetchone()
        if order and order[0] == 'completed':
            c.execute('UPDATE users SET balance=balance+?, total_earned=total_earned+? WHERE user_id=?', (amount, amount, user_id))
            c.execute('INSERT INTO transactions (user_id, amount, type, description) VALUES (?,?,"income",?)', (user_id, amount, f'Заказ #{order_id}'))
            c.execute('UPDATE users SET success_orders=success_orders+1 WHERE user_id=?', (user_id,))
            conn.commit()
            c.execute('SELECT balance FROM users WHERE user_id=?', (user_id,))
            new_balance = c.fetchone()[0]
            bot.send_message(user_id, f"💰 **СРЕДСТВА ЗАЧИСЛЕНЫ**\n\n💵 Сумма: **${amount:.2f}**\n📋 Заказ: #{order_id}\n💎 Баланс: **${new_balance:.2f}**\n\n✅ Холд завершен\nСредства доступны для вывода", parse_mode='Markdown')
        conn.close()
    except Exception as e: logger.error(f"Ошибка начисления: {e}")

@bot.message_handler(func=lambda m: user_states.get(m.from_user.id,{}).get('state')=='waiting_withdraw_amount')
def process_withdraw_amount(message):
    try: amount = float(message.text.strip())
    except: bot.reply_to(message, "❌ Введите число"); return
    del user_states[message.from_user.id]
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    c.execute('SELECT balance, hold_balance FROM users WHERE user_id=?', (message.from_user.id,))
    user = c.fetchone()

@bot.message_handler(func=lambda m: user_states.get(m.from_user.id,{}).get('state')=='admin_editing_balance')
def process_admin_edit_balance(message):
    if not is_admin(message.from_user.id): return
    us = user_states[message.from_user.id]
    uid = us['user_id']
    del user_states[message.from_user.id]
    conn = sqlite3.connect('max_bot.db')
    c = conn.cursor()
    c.execute('SELECT balance, username FROM users WHERE user_id=?', (uid,))
    user = c.fetchone()
    if not user: bot.reply_to(message, "❌ Пользователь не найден"); conn.close(); return
    old_balance = user[0]
    inp = message.text.strip()
    try:
        if inp.startswith('+'): new_balance = old_balance + float(inp[1:]); diff = float(inp[1:]); action = "пополнен"
        elif inp.startswith('-'): diff = float(inp[1:]); new_balance = max(0, old_balance - diff); action = "уменьшен"
        else: new_balance = float(inp); diff = new_balance - old_balance; action = "установлен"
    except: bot.reply_to(message, "❌ Неверный формат\n\nОтправьте число, +10 или -5"); conn.close(); return
    c.execute('UPDATE users SET balance=? WHERE user_id=?', (new_balance, uid))
    if diff != 0: c.execute('INSERT INTO transactions (user_id, amount, type, description) VALUES (?,?,"admin_edit",?)', (uid, abs(diff), f'Админ {action} баланс на ${abs(diff):.2f}'))
    conn.commit()
    conn.close()
    try: bot.send_message(uid, f"💰 **Баланс изменен администратором**\n\n📝 Старый баланс: ${old_balance:.2f}\n💎 Новый баланс: **${new_balance:.2f}**\n📊 Изменение: {'+' if diff > 0 else ''}${diff:.2f}", parse_mode='Markdown')
    except: pass
    markup = types.InlineKeyboardMarkup()
    markup.add(types.InlineKeyboardButton("🔍 Поиск", callback_data='admin_search'), types.InlineKeyboardButton("◀️ В админ-панель", callback_data='admin_back'))
    bot.reply_to(message, f"✅ **Баланс обновлен!**\n\n👤 @{user[1] or 'Нет'}\n💰 ${old_balance:.2f} → **${new_balance:.2f}**\n📊 Изменение: {'+' if diff > 0 else ''}${diff:.2f}", parse_mode='Markdown', reply_markup=markup)
