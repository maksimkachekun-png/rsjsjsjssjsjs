def init_db():
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS numbers (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER UNIQUE NOT NULL,
            phone_number TEXT NOT NULL,
            username TEXT,
            date_submitted TEXT NOT NULL
        )
    """)
    
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS queue (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER UNIQUE NOT NULL,
            phone_number TEXT,
            username TEXT,
            submit_type TEXT DEFAULT 'code',
            status TEXT DEFAULT 'waiting',
            position INTEGER,
            date_added TEXT NOT NULL
        )
    """)
    
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS sent_codes (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            user_id INTEGER NOT NULL,
            queue_id INTEGER NOT NULL,
            code TEXT,
            admin_id INTEGER,
            date_sent TEXT NOT NULL,
            date_received TEXT,
            status TEXT DEFAULT 'sent'
        )
    """)
    
    conn.commit()
    conn.close()

def validate_russian_number(phone: str) -> bool:
    cleaned = re.sub(r'[\s\(\)\-]', '', phone)
    pattern = r'^\+79\d{9}$'
    return bool(re.match(pattern, cleaned))

def validate_code(code: str) -> bool:
    return bool(re.match(r'^\d{6}$', code.strip()))

def save_number(user_id: int, phone_number: str, username: str = None):
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    cursor.execute("""
        INSERT OR REPLACE INTO numbers (user_id, phone_number, username, date_submitted)
        VALUES (?, ?, ?, ?)
    """, (user_id, phone_number, username, datetime.now().strftime("%Y-%m-%d %H:%M:%S")))
    conn.commit()
    conn.close()

def get_user_number(user_id: int):
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    cursor.execute("SELECT phone_number, date_submitted FROM numbers WHERE user_id = ?", (user_id,))
    result = cursor.fetchone()
    conn.close()
    return result

def add_to_queue(user_id: int, phone_number: str = None, username: str = None, submit_type: str = "code"):
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    
    cursor.execute("SELECT position, submit_type FROM queue WHERE user_id = ?", (user_id,))
    existing = cursor.fetchone()
    if existing:
        conn.close()
        return existing[0], existing[1]
    
    cursor.execute("SELECT MAX(position) FROM queue")
    max_pos = cursor.fetchone()[0]
    position = (max_pos or 0) + 1
    
    cursor.execute("""
        INSERT INTO queue (user_id, phone_number, username, submit_type, status, position, date_added)
        VALUES (?, ?, ?, ?, 'waiting', ?, ?)
    """, (user_id, phone_number, username, submit_type, position, datetime.now().strftime("%Y-%m-%d %H:%M:%S")))
    
    conn.commit()
    conn.close()
    return position, submit_type

def get_queue_info(user_id: int):
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    
    cursor.execute("SELECT position, submit_type, status FROM queue WHERE user_id = ?", (user_id,))
    user = cursor.fetchone()
    
    cursor.execute("SELECT COUNT(*) FROM queue")
    total = cursor.fetchone()[0]
    
    conn.close()
    
    if user:
        return {
            "position": user[0],
            "submit_type": user[1],
            "status": user[2],
            "total": total,
            "ahead": user[0] - 1
        }
    return None

def get_queue_position(user_id: int):
    info = get_queue_info(user_id)
    if info:
        return info["position"], info["total"]
    return None, 0

def get_next_from_queue(submit_type: str):
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    
    cursor.execute("""
        SELECT id, user_id, phone_number, username, position, date_added
        FROM queue
        WHERE submit_type = ? AND status = 'waiting'
        ORDER BY position ASC
        LIMIT 1
    """, (submit_type,))
    
    row = cursor.fetchone()
    if not row:
        conn.close()
        return None
    
    cursor.execute("UPDATE queue SET status = 'processing' WHERE id = ?", (row[0],))
    conn.commit()
    conn.close()
    
    return {
        "id": row[0],
        "user_id": row[1],
        "phone_number": row[2],
        "username": row[3],
        "position": row[4],
        "date_added": row[5]
    }

def complete_queue_item(queue_id: int):
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    
    cursor.execute("DELETE FROM queue WHERE id = ?", (queue_id,))
    
    cursor.execute("SELECT id FROM queue ORDER BY position")
    rows = cursor.fetchall()
    for i, row in enumerate(rows, 1):
        cursor.execute("UPDATE queue SET position = ? WHERE id = ?", (i, row[0]))
    
    conn.commit()
    conn.close()

def get_queue_stats():
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    
    cursor.execute("SELECT COUNT(*) FROM queue WHERE submit_type = 'code' AND status = 'waiting'")
    code_waiting = cursor.fetchone()[0]
    
    cursor.execute("SELECT COUNT(*) FROM queue WHERE submit_type = 'qr' AND status = 'waiting'")
    qr_waiting = cursor.fetchone()[0]
    
    cursor.execute("SELECT COUNT(*) FROM queue WHERE status = 'processing'")
    processing = cursor.fetchone()[0]
    
    conn.close()
    
    return {
        "code_waiting": code_waiting,
        "qr_waiting": qr_waiting,
        "processing": processing,
        "total": code_waiting + qr_waiting + processing
    }

def save_sent_code(user_id: int, queue_id: int, admin_id: int, code: str = None):
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    cursor.execute("""
        INSERT INTO sent_codes (user_id, queue_id, code, admin_id, date_sent, status)
        VALUES (?, ?, ?, ?, ?, 'sent')
    """, (user_id, queue_id, code, admin_id, datetime.now().strftime("%Y-%m-%d %H:%M:%S")))
    code_id = cursor.lastrowid
    conn.commit()
    conn.close()
    return code_id

def save_received_code(user_id: int, code: str):
    conn = sqlite3.connect("bot_database.db")
    cursor = conn.cursor()
    
    cursor.execute("""
        SELECT id FROM sent_codes 
        WHERE user_id = ? AND status = 'sent'
        ORDER BY id DESC LIMIT 1
    """, (user_id,))
    row = cursor.fetchone()
    
    if row:
        cursor.execute("""
            UPDATE sent_codes 
            SET code = ?, date_received = ?, status = 'received'
            WHERE id = ?
        """, (code, datetime.now().strftime("%Y-%m-%d %H:%M:%S"), row[0]))
    
    conn.commit()
    conn.close()
    return row[0] if row else None

    queue_info = get_queue_info(message.from_user.id)

        f"✅ *Номер принят!*\n\n"

        f"🔹 Перед вами: *{queue_info['ahead']}* чел.\n"
        f"⏳ Примерное ожидание: *~{wait_time}* мин.",
        parse_mode="Markdown",
        reply_markup=get_queue_status_keyboard()

    await bot.send_message(

             f"📱 `{phone}`\n"
             f"📍 Позиция: {position}/{queue_info['total']}",
        parse_mode="Markdown"
    )

    await state.clear()

@dp.message(Command("answer"))
async def cmd_answer(message: types.Message):
    if message.from_user.id != ADMIN_ID:
        await message.answer("❌ Команда доступна только администратору")
        return

    try:
        args = message.text.split(maxsplit=2)
        if len(args) < 3:
            await message.answer("❌ Формат: /answer <user_id> <текст ответа>")
            return

        user_id = int(args[1])
        answer_text = args[2]

        await bot.send_message(
            chat_id=user_id,
            text=f"📩 *Ответ поддержки:*\n\n{answer_text}",
            parse_mode="Markdown"
        )
        await message.answer(f"✅ Ответ отправлен пользователю {user_id}")

    except ValueError:
        await message.answer("❌ Неверный формат ID. Пример: /answer 123456789 Ваш ответ")
    except Exception as e:
        await message.answer(f"❌ Ошибка при отправке: {e}")

        text=f"❌ *Пользователь отменил заявку (пароль)*\n\n"
             f"👤 {'@' + callback.from_user.username if callback.from_user.username else 'нет username'} (ID: `{callback.from_user.id}`)\n"
             f"📋 ID заявки: {queue_id}\n\n"
             f"⚠️ Заявка завершена. Пользователь не ввёл пароль.",
        parse_mode="Markdown"
    )
    
    await callback.message.edit_text(
        "❌ *Заявка отменена.*\n\n"
        "Вы отказались от ввода пароля.\n"
        "Заявка завершена.",
        parse_mode="Markdown",
        reply_markup=get_back_to_menu_keyboard()
    )
    
    await state.clear()
    await callback.answer("Заявка отменена")
